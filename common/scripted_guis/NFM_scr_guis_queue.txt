
scripted_gui={
	NFM_Open_NFqueue={
		context_type=national_focus_context
		window_name="NFM_OpenNFqueue"
		parent_window_name="nationalfocusview"
		dirty=NFM_d
		ai_enabled={always=no}
		visible={tag=root NFM_allowed=yes}
		effects={
			NFM_OpenButton_click={
				NFM_dirty=yes
				if ={
					limit={has_country_flag=NFM_OpenNFqueueFlag}
					clr_country_flag=NFM_OpenNFqueueFlag
					clr_country_flag=NFM_OpenTemplateFlag
				}
				else={set_country_flag=NFM_OpenNFqueueFlag}
			}
		}
	}
	NFM_NFqueue_UIs={
		context_type=player_context
		window_name="NFM_queue_window"
		parent_window_name="nationalfocusview"
		#dirty=NFM_d
		ai_enabled={always=no}
		visible={tag=root has_country_flag=NFM_OpenNFqueueFlag}
		triggers={
			NFM_queue_up_click_enabled={check_variable={i>0}}
			NFM_queue_down_click_enabled={
				set_temp_variable={ii=i}
				add_to_temp_variable={ii=1}
				check_variable={ii<NFM_focus_queue^num}
			}
			NFM_template_button_visible={
				#NFM_has_template=yes
				has_variable=NFM_templates^0
				date<1936.1.2
			}
		}
		properties={NFM_NFicon={image=[NFM_focusicon]}}
		effects={
			NFM_close_button_click={
				clr_country_flag=NFM_OpenNFqueueFlag
				clr_country_flag=NFM_OpenTemplateFlag
				NFM_dirty=yes
			}
			NFM_clear_button_click={
				NFM_del_all=yes
			}
			NFM_template_button_click={
				if={
					limit={has_country_flag=NFM_OpenTemplateFlag}
					clr_country_flag=NFM_OpenTemplateFlag
				}
				else={set_country_flag=NFM_OpenTemplateFlag}
			}
			NFM_remove_from_queue_button_click={
				set_variable={n=v} NFM_set=yes
			}
			NFM_queue_up_click={
				set_temp_variable={ti=-1}
				NFM_list_swap_in_queue=yes
			}
			NFM_queue_down_click={
				set_temp_variable={ti=1}
				NFM_list_swap_in_queue=yes
			}
			NFM_queue_up_shift_click={
				set_temp_variable={tv=v}
				for_each_loop={
					array=NFM_focus_queue value=vv index=ii break=b
					set_variable={NFM_focus_queue^ii=tv}
					set_temp_variable={tv=vv}
					if={
						limit={check_variable={ii=i}}
						set_temp_variable={b=1}
					}
				}
				clr_country_flag=NFM_error
				NFM_set_text_variables=yes
			}
			NFM_queue_down_shift_click={
				add_to_array={NFM_focus_queue=v}
				remove_from_array={array=NFM_focus_queue index=i}
				clr_country_flag=NFM_error
				NFM_set_text_variables=yes
			}
		}
		dynamic_lists={
			NFM_queue_list={
				array=NFM_focus_queue
				country_scope_entry_container="NFM_NF_item"
				change_scope=no
				value=v index=i
			}
		}
	}
	NFM_NFitem_UIs={
		context_type=player_context
		window_name="NFM_NF_item"
		#dirty=NFM_d
		visible={always=no}
		ai_enabled={always=no}
	}
	NFM_Template_UIs={
		context_type=player_context
		window_name="NFM_template_window"
		parent_window_name="nationalfocusview"
		#dirty=NFM_d
		ai_enabled={always=no}
		visible={date<1936.1.2 tag=root has_country_flag=NFM_OpenTemplateFlag}
		properties={NFM_templateicon={image=[NFM_template_icon]}}
		effects={
			NFM_close_button_click={
				clr_country_flag=NFM_OpenTemplateFlag
			}
			NFM_template_button_click={
				NFM_load_template_queue=yes
			}
		}
		dynamic_lists={
			NFM_template_list={
				array=NFM_templates
				country_scope_entry_container="NFM_template_item"
				change_scope=no
				value=tv index=ti
			}
		}
	}
	NFM_templateitem_UIs={
		context_type=player_context
		window_name="NFM_template_item"
		#dirty=NFM_d
		visible={always=no}
		ai_enabled={always=no}
	}
}
