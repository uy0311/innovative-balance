NFM_try_load_oob={
	if={limit={has_country_flag=NFM_NFMactivated}
		meta_effect = {
			text={set_variable={f[nfid]=1} set_variable={t[nfid]=0}}
			nfid="[?NFM_nfnum]"
		}
		set_variable={NFM_nfnum=0}
		set_country_flag=NFM_error
		clr_country_flag=NFM_NFMactivated
		NFM_dirty=yes
	}
	else={
		if={
			limit={has_variable=NFM_focus_queue^0}
			NFM_load_oob=yes
		}
	}
}
NFM_set={
	meta_effect={
		text={
			if={limit={check_variable={f[nfid]=1}}
				add_to_array={NFM_focus_queue=[nfid]}
				set_variable={f[nfid]=2}
				set_variable={t[nfid]=NFM_focus_queue^num}
			}
			else_if ={
				limit={check_variable={f[nfid]=2}}
				remove_from_array={NFM_focus_queue=[nfid]}
				set_variable={f[nfid]=1}
				set_variable={t[nfid]=0}
			}
		}
		nfid="[?n]"
	}
	clr_country_flag=NFM_error
	NFM_set_text_variables=yes
}
NFM_set_text_variables={
	for_each_loop={
		array=NFM_focus_queue value=vv index=ii
		meta_effect={
			text={
				add_to_temp_variable={ii=1}
				set_variable={t[nfid]=ii}
			}
			nfid="[?vv]"
		}
	}
	NFM_dirty=yes
}
NFM_sync_tree_to_queue={
	for_each_loop={
		array=NFM_focus_queue value=vv index=ii
		meta_effect={
			text={
				add_to_temp_variable={ii=1}
				set_variable={t[nfid]=ii}
				set_variable={f[nfid]=2}
			}
			nfid="[?vv]"
		}
	}
	NFM_dirty=yes
}
NFM_load_oob={
	set_variable={NFM_nfnum=NFM_focus_queue^0}
	meta_effect={
		text={
			if={limit={[tag]_set[nfid]=yes}
			remove_from_array={array=NFM_focus_queue index=0}
			set_country_flag=NFM_NFMactivated
			clr_country_flag=NFM_error
			load_oob=[tag]/[tag]_[nfid]
			if={limit={is_in_array={NFM_seleffs=[nfid]}}
				NFM_load_seleff=yes
			}
			set_variable={f[nfid]=4}
			set_variable={t[nfid]=0}
			}
			else={set_variable={NFM_nfnum=0} set_country_flag=NFM_error}
		}
		nfid="[?NFM_nfnum]"
		#tag="[root.GetTag]"
		tag="[NFM_TAG_with_shared]"
	}
	set_variable={v=NFM_nfnum} #queue loc
	NFM_set_text_variables=yes
}
NFM_load_seleff={
	meta_effect={
		text={[tag]_seleff[nfid]=yes}
		nfid="[?NFM_nfnum]"
		tag="[NFM_TAG_with_shared]"
	}
}
